~zig-emacs~ provides idiomatic, high-performance Zig bindings for GNU Emacs [[https://www.gnu.org/software/emacs/manual/html_node/elisp/Writing-Dynamic-Modules.html][dynamic modules]].

* Features

- *Zero Header Setup*: Automatically translates bundled ~include/emacs-module.h~ via Zig's ~addTranslateC~ during build.
- *Ergonomic Function Signatures*: Functions accept native Zig scalar types (~i64~, ~bool~, ~f64~, ~?T~) with automatic argument unpacking, type checking (~wrong-type-argument~), and auto-converted return values.
- *Struct & Plist/Alist Serialization*: Seamless bidirectional mapping between Zig structs and Elisp property lists / association lists via ~fromPlist~ (with default field value support) and ~makePlist~/~makeAlist~.
- *Type-safe User Pointers*: Wrap arbitrary Zig structs with typed helpers and automatic finalizers tied to Emacs GC.
- *Non-blocking Async Streaming & IO*: Stream native results via ~openChannel~, with zero-allocation ~channel.print(...)~ and standard Zig 0.17 ~std.Io.Writer~ integration.
- *Scoped Arena Memory Management*: Lightweight ~env.arena(gpa)~ helper to eliminate tedious manual freeing of temporary allocations.
- *Idiomatic Error Handling*: Seamless mapping between Zig error unions and Emacs non-local exits (~signal~ and ~throw~).

* Requirements

- Zig 0.17.0 or newer
- GNU Emacs 28 or newer

* Quickstart

Every module must call ~moduleInit~ at comptime and provide an ~init~ function. Registered functions accept ~emacs.Env~ as their first argument and can declare either raw ~emacs.Value~ or native Zig scalar types:

#+begin_src zig
const std = @import("std");
const emacs = @import("emacs");

comptime {
    emacs.moduleInit(@This());
}

// Functions can declare native Zig types directly!
// zig-emacs handles automatic unpacking, type checks, and return conversions.
fn add(_: emacs.Env, lhs: i64, rhs: i64) i64 {
    return lhs + rhs;
}

pub fn init(env: emacs.Env) c_int {
    env.makeFunction(
        "zig-add",
        add,
        .{
            .doc_string = "Add two integers.",
            .interactive_spec = "nFirst number: \nnSecond number: ",
        },
    ) catch return 1;

    return 0;
}
#+end_src

Build the module and load it in Emacs:

#+begin_src emacs-lisp
(add-to-list 'load-path "/path/to/zig-emacs/zig-out/lib/")
(load "libzig-example")
(zig-add 1 2) ;; => 3
#+end_src

* Core Guides

** Ergonomic Function Signatures & Automatic Conversions

Functions registered via ~makeFunction~ do not need to deal with low-level ~emacs.Value~ unwrapping. You can declare native Zig types directly, and ~zig-emacs~ will validate arguments, raise standard ~(wrong-type-argument PREDICATE VALUE)~ errors on mismatches, and convert return values:

#+begin_src zig
// 1. Native arguments and return values (with optional parameters)
fn calculate(_: emacs.Env, a: i64, b: i64, bonus: ?i64) i64 {
    return a * 10 + b + (bonus orelse 0);
}

// 2. Composing Elisp lists and vectors from heterogeneous tuples
fn buildResponse(env: emacs.Env) !emacs.Value {
    return env.list(.{
        @as(i64, 200),
        "status OK",
        true,
    });
}
#+end_src

** Plist, Alist & Struct Deserialization

~zig-emacs~ provides reflection-powered serialization between native Zig structs and Elisp property lists / association lists:

#+begin_src zig
const ServerConfig = struct {
    host: []const u8 = "localhost",
    port: i32 = 8080,
    verbose: bool = false,
    token: ?[]const u8 = null,
};

// Deserializing an Elisp plist with default values:
fn configure(env: emacs.Env, plist_val: emacs.Value) !emacs.Value {
    var arena = env.arena(std.heap.smp_allocator);
    defer arena.deinit();

    const config = try env.fromPlist(ServerConfig, plist_val, arena.allocator());
    // config.host, config.port, etc. are populated with defaults if omitted in plist!
    return env.boolean(config.verbose);
}

// Serializing a Zig struct into an Elisp plist or alist:
fn exportConfig(env: emacs.Env) !emacs.Value {
    return env.makePlist(.{
        .name = "my-service",
        .port = @as(i64, 9090),
        .active = true,
    });
}
#+end_src

** Allocator Context & Scoped Arena

Emacs ~Env~ does not manage memory allocators. Pass application state via ~FuncContext.user_data~ and use ~env.arena~ for hassle-free temporary memory management:

#+begin_src zig
const ModuleContext = struct {
    gpa: std.mem.Allocator,
};

var module_context: ModuleContext = .{
    .gpa = std.heap.smp_allocator,
};

fn greeting(env: emacs.Env, value: emacs.Value) !emacs.Value {
    const context = env.getUserData(ModuleContext) orelse return error.MissingModuleContext;

    // Scoped arena automatically frees all temporary allocations when deinit is called
    var arena = env.arena(context.gpa);
    defer arena.deinit();

    const text = try env.extractString(arena.allocator(), value);
    return env.makeString(text);
}

pub fn init(env: emacs.Env) c_int {
    env.makeFunction(
        "zig-greeting",
        greeting,
        .{ .user_data = @ptrCast(&module_context) },
    ) catch return 1;
    return 0;
}
#+end_src

** Typed User Pointers

Bind native state directly to Emacs Lisp values with optional garbage-collection finalizers:

#+begin_src zig
const Resource = struct {
    gpa: std.mem.Allocator,

    fn deinit(self: *Resource) void {
        self.gpa.destroy(self);
    }
};

fn makeResource(env: emacs.Env, gpa: std.mem.Allocator) !emacs.Value {
    const resource = try gpa.create(Resource);
    resource.gpa = gpa;
    return env.makeUserPointer(Resource, resource, Resource.deinit);
}

fn useResource(env: emacs.Env, value: emacs.Value) emacs.Value {
    const resource = env.getUserPointer(Resource, value) orelse return env.nil;
    _ = resource;
    return env.t;
}
#+end_src

Pass ~null~ as the finalizer callback for unmanaged or borrowed pointers.

** Calling Emacs Functions & Error Handling

Invoke any Emacs Lisp function with ~Env.funcall~, which automatically detects signals and throws:

#+begin_src zig
var args = [_]emacs.Value{env.makeString("Hello from Zig!")};
const result = try env.funcall("message", &args);
#+end_src

Convenience wrappers are available:
- ~env.message(str)~: Print directly to the echo area / ~*Messages*~ buffer.
- ~env.reportError(prefix, err)~: Call the Emacs ~error~ function.
- ~env.signal(err, args)~: Raise an Emacs signal from a Zig error.
- ~env.signalSymbol(symbol_name, data)~: Signal a specific interned error condition.

** Native Types, Sequences & Object Wrappers

~zig-emacs~ provides rich, object-oriented Zig structs for manipulating Emacs data structures without low-level C-ABI boilerplate:

#+begin_src zig
// 1. Pattern matching on Emacs value types
switch (env.getType(val)) {
    .integer => { const n = env.extractInteger(val); },
    .string => { const s = try env.extractString(allocator, val); },
    .nil => { ... },
    else => { ... },
}

// 2. Unified Sequence (accepts both Emacs Vector and List!)
if (env.toSequence(val)) |seq| {
    const len = try seq.len();
    const items = try seq.toSlice(allocator);
    defer allocator.free(items);
}

// 3. Object-oriented Vector
const vec = try emacs.Vector.create(env, &[_]emacs.Value{ a, b });
const first_item = vec.get(0);
vec.set(1, c);

// 4. Object-oriented List
const list = try emacs.List.create(env, &[_]emacs.Value{ a, b });
const head = try list.car();
const tail = try list.cdr();

// 5. Type-safe UserPointer handle
const MyHandle = emacs.UserPointer(MyResource);
const handle = MyHandle.create(env, res, MyResource.deinit);
// Or wrap an existing user pointer value:
// const handle = MyHandle.init(env, user_ptr_val);
if (handle.getChecked(MyResource.magic)) |res_ptr| {
    res_ptr.doWork();
}
handle.deinit(); // sets user-ptr to null
#+end_src

** Non-blocking Async Streaming & Channel IO

~openChannel~ binds native code to the write end of an Emacs ~make-pipe-process~. This allows background threads to stream results without locking the main Emacs UI:

#+begin_src emacs-lisp
(let ((process
       (make-pipe-process
        :name "zig-stream"
        :buffer nil
        :filter (lambda (_proc chunk)
                  (message "Received: %s" chunk)))))
  (zig-start-stream process))
#+end_src

On the Zig side, ~Channel~ provides zero-allocation stack formatting (~channel.print~) and seamless adaptation to the standard Zig 0.17 ~std.Io.Writer~:

#+begin_src zig
const Worker = struct {
    channel: emacs.Channel,

    fn run(self: *Worker) void {
        defer {
            var ch = self.channel;
            ch.deinit() catch {};
        }

        // 1. Zero-allocation stack formatted print
        self.channel.print("Processed item {d} of {d}\n", .{ 1, 10 }) catch return;

        // 2. Standard Zig 0.17 Writer adapter
        var buf: [512]u8 = undefined;
        var adapter = self.channel.writer(&buf);
        const writer = &adapter.writer;
        writer.print("Stream payload: {s}\n", .{"ready"}) catch return;
        writer.flush() catch return;
    }
};

fn startStream(env: emacs.Env, proc_val: emacs.Value) !emacs.Value {
    const channel = try env.openChannel(proc_val);
    // Move channel into background worker thread...
    return env.t;
}
#+end_src
